describe('Inspecting an evaluation', () => {
  it('defaults to a denied evaluation and shows context, mismatch and decision', async () => {
    renderAt('/assurance')
    await ready()
    expect(document.querySelector('[data-inspector]')).toBeInTheDocument()
    expect(screen.getByRole('status', { name: 'Intent-action result' })).toBeInTheDocument()
  })

  it('HLT-02: highlights the Patient A / Patient B discrepancy and keeps decision, gateway and execution apart', async () => {
    renderAt('/assurance?eval=run-HLT-02-protected-pe2')
    await ready()
    const result = await screen.findByRole('status', { name: 'Intent-action result' })
    expect(result).toHaveTextContent('Mismatch found')
    expect(result).toHaveTextContent('The request concerns Patient A record and Visit summary for Patient A, but the action targets Patient B record.')
    expect(document.querySelector('[data-verification="mismatch"]')).toBeInTheDocument()
    expect(document.querySelectorAll('[data-check="mismatch"]').length).toBeGreaterThanOrEqual(3)
    const inspector = document.querySelector('[data-inspector]') as HTMLElement
    expect(within(inspector).getAllByText('DENY').length).toBeGreaterThan(0)
    expect(within(inspector).getByText('Blocked — not executed. The resource was left unchanged.')).toBeInTheDocument()
    expect(within(inspector).getByText('Not executed')).toBeInTheDocument()
    expect(within(inspector).getByText(/An ALLOW is never proof that anything ran/)).toBeInTheDocument()
    expect(inspector.querySelector('[data-field="simulation-id"]')).toHaveTextContent('run-HLT-02-protected')
    expect(within(inspector).getByText('POL-SCOPE-001', { exact: false })).toBeInTheDocument()
  })

  it('ALLOW is shown as a decision, not as proof of execution', async () => {
    renderAt('/assurance?eval=run-FIN-01-protected-pe1')
    await ready()
    const inspector = await waitForInspector()
    expect(within(inspector).getAllByText('The action meets the demonstration rules.').length).toBeGreaterThan(0)
    expect(within(inspector).getByText(/never proof that anything ran/)).toBeInTheDocument()
    expect(inspector.querySelector('[data-stage="2 · Gateway outcome"]')).toBeInTheDocument()
  })
